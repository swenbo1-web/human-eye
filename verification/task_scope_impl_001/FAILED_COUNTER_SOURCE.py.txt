def on_surface(point,mesh):
    p=np.asarray(point)
    for tri in mesh.vertices[mesh.faces]:
        u,v,w=tri[1]-tri[0],tri[2]-tri[0],p-tri[0]
        n=np.cross(u,v);length=np.linalg.norm(n)
        if not length or abs(np.dot(n,w))/length>1e-9:continue
        uu,vv,uv,wu,wv=map(float,(np.dot(u,u),np.dot(v,v),np.dot(u,v),np.dot(w,u),np.dot(w,v)))
        determinant=uu*vv-uv*uv
        if not determinant:continue
        a,b=(wu*vv-wv*uv)/determinant,(wv*uu-wu*uv)/determinant
        if a>=-1e-9 and b>=-1e-9 and a+b<=1+1e-9:return True
    return False

    def test_original_geometry_three_states_and_old_pair_equivalence_wire(self):
        truth=json.loads(original.TRUTH.read_bytes());records=[]
        meshes=original.snapshot_meshes('initial');self.arrange(list(meshes.values()))
        for state in ('initial','unchanged','changed'):
            if state=='changed':
                meshes=original.snapshot_meshes('changed')
                self.host.scene['meshes']=[mesh.to_dict() for mesh in meshes.values()]
            q={'mode':'task','target':{'id':'Object_01'},'candidates':[{'id':'Object_0'+n} for n in ('2','3','4')],
               'requested_facts':['aabb','closed_solid_relation','surface_proximity']}
            response,value=self.call(q);set_work=self.work();expected=truth['states'][state]
            self.assertFalse(response['result']['isError'])
            self.assertEqual(value['summary']['penetrating_ids'],[k for k,p in expected['probes'].items() if p['relation']=='penetrating'])
            self.assertEqual(value['summary']['nearest_separated_ids'],[expected['nearest_separated_probe']])
            old_bytes=0
            for row in value['pairs']:
                oid=row['object_id'];pair_query={'mode':'task','objects':[{'id':'Object_01'},{'id':oid}],
                    'requested_facts':['aabb','closed_solid_relation','surface_proximity']}
                old_response,old_value=self.call(pair_query);old_bytes+=len(wire.encode_response(old_response).encode())
                self.assertEqual(row['facts']['closed_solid_relation'],old_value['facts']['closed_solid_relation'])
                left,right=row['facts']['surface_proximity'],old_value['facts']['surface_proximity']
                for key in ('status','witness_status','direction_status'):self.assertEqual(left[key],right[key])
                self.assertEqual(left['value']['distance_m'],right['value']['distance_m'])
                self.assertEqual(left['value']['topology'],right['value']['topology'])
                for fact in (left,right):
                    witness=fact['value']['nearest_witness']
                    self.assertTrue(on_surface(witness['a'],meshes['Object_01']))
                    self.assertTrue(on_surface(witness['b'],meshes[oid]))
                    self.assertAlmostEqual(math.dist(witness['a'],witness['b']),fact['value']['distance_m'],delta=1e-9)
                if expected['probes'][oid]['relation']=='separated':
                    low,high=expected['probes'][oid]['distance_interval_m'];gap=row['separation_gap']['value'];self.assertLessEqual(low,gap);self.assertLessEqual(gap,high)
            records.append({'state':state,'set_full_JSONRPC_bytes':len(wire.encode_response(response).encode()),
                            'three_pair_full_JSONRPC_bytes':old_bytes,'set_work':set_work,'result':value})
