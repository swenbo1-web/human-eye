def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def compact(o):
    return json.dumps(o, ensure_ascii=False, separators=(',', ':'), allow_nan=False).encode('utf8')

def dist(v):
    return {'count': len(v), 'minimum': min(v) if v else None, 'median': statistics.median(v) if v else None,
            'maximum': max(v) if v else None, 'raw_samples': v}
