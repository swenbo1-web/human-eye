"""Copy only delivered fixed Git blobs; never run truth/product/Blender code."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess

HERE = Path(__file__).resolve().parent
SOURCE_REPOSITORY = Path('C:/Users/Administrator/.codex/worktrees/m4-single-pass-acquisition/hefe')
SOURCE_COMMIT = 'b9336a8037d4781584b703e1aad8f24845f0a5b2'
SOURCE_PREFIX = 'verification/consumption_prep_001/'
HELPER = Path('D:/hefe/verification/fixes/20261106_perf010_r2/installed/verification/visual004_native.py')
HELPER_SHA = '5bc1152aeafab89220c93edeec51cbe32ace458cc07d3e7d0507b6412b816fb3'
TASK_ORDER = [('T01','N'),('T01','P'),('T02','P'),('T02','N'),
              ('T03','N'),('T03','P'),('T04','P'),('T04','N')]


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def git_blob(path):
    return subprocess.check_output(['git','show',SOURCE_COMMIT+':'+path],cwd=SOURCE_REPOSITORY)


def write(path, raw):
    path.parent.mkdir(parents=True,exist_ok=True)
    if path.exists():
        assert path.read_bytes()==raw, 'do_not_replace_changed_material:'+str(path)
    else:
        path.write_bytes(raw)


def main():
    names = subprocess.check_output(['git','ls-tree','-r','--name-only',SOURCE_COMMIT,
        SOURCE_PREFIX],cwd=SOURCE_REPOSITORY,text=True).splitlines()
    assert len(names)==18 and all(name.startswith(SOURCE_PREFIX) for name in names)
    rows=[]
    for name in names:
        raw=git_blob(name)
        destination=HERE/'frozen'/name[len(SOURCE_PREFIX):]
        write(destination,raw)
        rows.append(dict(source_path=name,local_path=destination.relative_to(HERE).as_posix(),
            bytes=len(raw),sha256=digest(raw)))
    assets_name='verification/m4_capture_qa_001/run_01/assets/ASSETS.json'
    assets_raw=git_blob(assets_name)
    write(HERE/'admin/ORIGINAL_ASSETS.json',assets_raw)
    assets=json.loads(assets_raw)
    source_assets=json.loads((HERE/'frozen/admin/ASSET_INDEX.json').read_bytes())
    prompts=json.loads((HERE/'frozen/consumer_prompts.json').read_bytes())
    seal=json.loads((HERE/'frozen/DELIVERY_MANIFEST.json').read_bytes())
    sealed={row['path']:row for row in seal['files']}
    for row in rows:
        relative=row['local_path'][len('frozen/'):]
        if relative=='DELIVERY_MANIFEST.json':continue
        assert sealed[relative]['sha256']==row['sha256']
        assert sealed[relative]['bytes']==row['bytes']
    tasks=[]
    assert [p['task_id'] for p in prompts['tasks']]==['T01','T02','T03','T04']
    assert source_assets['assets_manifest_sha256']==digest(assets_raw)
    for prompt in prompts['tasks']:
        task=prompt['task_id']
        raw=prompt['prompt'].encode('utf8')
        assert digest(raw)==prompt['prompt_utf8_sha256']
        prompt_path=HERE/'prompts'/(task+'.txt')
        write(prompt_path,raw)
        metadata=next(row for row in source_assets['tasks'] if row['task_id']==task)
        original=next(row for row in assets['cases'] if row['case']==metadata['case'])
        asset=Path(metadata['asset_path'])
        asset_git_path=asset.relative_to(SOURCE_REPOSITORY).as_posix()
        asset_blob=git_blob(asset_git_path)
        assert digest(asset_blob)==digest(asset.read_bytes())==metadata['asset_sha256']==original['sha256']
        assert hashlib.sha256(json.dumps(original['recipe'],sort_keys=True).encode()).hexdigest()==metadata['recipe_sha256']
        tasks.append(dict(task_id=task,prompt_path=prompt_path.relative_to(HERE).as_posix(),
            prompt_utf8_sha256=digest(raw),prompt_utf8_bytes=len(raw),prompt_has_final_newline=raw.endswith(b'\n'),
            source_asset_path=str(asset),asset_git_path=asset_git_path,asset_sha256=metadata['asset_sha256'],
            asset_bytes=len(asset_blob),source_recipe_sha256=metadata['recipe_sha256'],
            asset_prep_copied=False,controller_only_asset_metadata=True,
            future_consumer_asset_name='input.blend',future_copy_rule='private NativeSession workspace only; exact source bytes; no case filenames in prompt')))
    helper_raw=HELPER.read_bytes()
    assert digest(helper_raw)==HELPER_SHA
    write(HERE/'admin/visual004_native.py',helper_raw)
    ref_paths=['docs/HUMAN_EYE_FRAMEWORK_REFACTOR_PLAN_FINAL.md',
        'docs/REGION_WITNESS_TASK_PILOT.md','docs/V21_POST_ANCHOR_CONSUMPTION_TRIAL.md',
        'verification/fixes/evidence_consume_001/run_01/RESULTS.md']
    source_index=json.loads((HERE/'frozen/SOURCE_INDEX.json').read_bytes())
    ref_rows=[]
    for path in ref_paths:
        raw=git_blob(path)
        expected=next(row for row in source_index['documents'] if row['path'].replace('\\','/').endswith('/'+path))
        assert digest(raw)==expected['sha256']
        destination=HERE/'admin/references'/path
        write(destination,raw)
        ref_rows.append(dict(source_path=path,local_path=destination.relative_to(HERE).as_posix(),
            sha256=digest(raw),bytes=len(raw),consumer_visible=False))
    manifest=dict(cycle_id='CONSUMPTION-RUN-PREP-001',extracted_utc=datetime.now(timezone.utc).isoformat(),
        source_repository=str(SOURCE_REPOSITORY),source_commit=SOURCE_COMMIT,
        source_read_mechanism='git show fixed commit:path; no other branch checkout/index reads',
        frozen_prep_files=rows,original_assets_manifest=dict(source_path=assets_name,
            local_path='admin/ORIGINAL_ASSETS.json',sha256=digest(assets_raw),bytes=len(assets_raw)),
        tasks=tasks,cell_order=[dict(ordinal=i+1,task_id=task,arm=arm) for i,(task,arm) in enumerate(TASK_ORDER)],
        references=ref_rows,neutral_helper=dict(source_path=str(HELPER),local_path='admin/visual004_native.py',
            sha256=HELPER_SHA,bytes=len(helper_raw),imported_or_executed=False),
        historical_consumer_settings=dict(model_alias='gpt-6.1-sol',reasoning_effort='ultra',
            source_path='admin/references/verification/fixes/evidence_consume_001/run_01/RESULTS.md',
            source_line=21,actual_backend='unexposed',current_available_model='unverified_by_material_extraction'),
        consumer_visibility=dict(allow='one exact prompts/Txx.txt content plus current approved tools and runtime-owned neutral asset name only',
            deny='frozen/*; admin/*; MATERIAL_MANIFEST.json; source case filenames; old raw; truth; scoring; arm/order map; old results; helper implementation',
            hard_file_isolation=False,
            limitation='Arbitrary full Python permits filesystem/network access of the caller account; excluded from automatic injection is not inaccessible. Public calls must be audited; do not claim blind trial.'),
        actual_activity=dict(model_answers=0,Blender_sessions=0,native_RPC=0,product_RPC=0,API_calls=0,
            product_imports=0,truth_algorithm_execution=0,asset_copies=0))
    (HERE/'MATERIAL_MANIFEST.json').write_text(json.dumps(manifest,indent=2,ensure_ascii=False)+'\n',encoding='utf8')
    print(json.dumps(dict(extracted_prep_blobs=len(rows),tasks=len(tasks),asset_hashes_verified=4,
        helper_sha256=HELPER_SHA,model_answers=0,Blender_sessions=0,RPC=0,product_imports=0)))


if __name__=='__main__':main()
